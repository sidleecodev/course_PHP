5.後臺管理員功能設置

終端機 php artisan make:controller Admin/ManagerController

自動生成檔案
app/Http/Controllers/Admin/ManagerController.php

輸入以下內容：
（請留意有沒有出現use App\Models\Admin\Manager;）
<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\Admin\Manager;
use Illuminate\Http\Request;

class ManagerController extends Controller
{
    public function list(){
        // get: 取得全部資料，也可以用all -> Manager::all();
        // 將取得的資料以$list變數儲存
        // 此處Manager是一個資料表，在app\Models\Admin\Manager.php
 	$list = Manager::paginate(10);
	/*
	    原本是指令是這個，但後續要做分頁(9.)，所以改為上面這個，括號內為幾筆為一頁
            $list = Manager::get();
	*/
        // compact: 將資料渲染到admin.manager.list這個網頁（此時list還沒創建）
        // compact("list"): 這個list為上一行的$list，兩個名稱要一樣
        return view("admin.manager.list", compact("list"));
    }
}

7.後臺管理功能要新增路由
routes/web.php

Route::group(["prefix" => "admin"], function () {
    // 後臺會員登入
    Route::get("/", [AdminController::class, "index"]);
    // 表單送到後臺
    Route::post("login", [AdminController::class, "login"]);
    // 登入成功跳轉頁面，在跳轉之前先跑一次middleware，確保有登入過才跳轉頁面
    Route::get("home", [AdminController::class, "home"])->middleware("manager");

    // !!!主要新增這裡!!!
    Route::middleware("manager")->group(function(){
        // 網址: /admin/manager
        Route::prefix("manager")->group(function(){
            // 網址: /admin/manager/list
            Route::get("list", [ManagerController::class, "list"]);
        });
    });
});

~~附加總筆數的功能
ManagerController.php修改為
<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\Admin\Manager;
use Illuminate\Http\Request;

class ManagerController extends Controller
{
    public function list(){
        // get: 取得全部資料，也可以用all -> Manager::all();
        // 將取得的資料以$list變數儲存
        // 此處Manager是一個資料表，在app\Models\Admin\Manager.php
        $list = Manager::paginate(10);
	// !!主要修改這邊
        // 計算總筆數
        $total = Manager::count();

        // compact: 將資料渲染到admin.manager.list這個網頁
        // compact("list"): 這個list為上一行的$list，兩個名稱要一樣
        return view("admin.manager.list", compact("list", "total"));
	// 到這邊!!
    }
}
