2. 新增前臺大事記控制器：php artisan make:controller Front/FrontMileStoneController
加入list function：
    public function list(){
        // distinct: 去除重複資料
        // pluck: 取出指定欄位的值
        // ->orderByDesc("years"): 從大到小，也可以用orderby("years", "desc")
        $years = MileStone::distinct()->orderByDesc("years")->pluck("years");
        
        /* 
        也可以寫成這樣，這種查詢會包含欄位名稱，取值時要加上->，例如： $years[0]->years
        如果用pluck，則取值用$years[0]
        $years = MileStone::selectRaw("years")->distinct()->orderByDesc("years");

        範例
        foreach($years as $data){
            echo $data->years;
            // 如果是用pluck，則直接取值echo $data
        }
        */

        $list = MileStone::get();
        return view("front.milestone.list", compact("list", "years"));
    }
