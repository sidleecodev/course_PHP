<script>
    // 傳入被點擊的按鈕，刪除成功後移除它所在的資料列
    function doDelete(id, btn) {
        Swal.fire({
            icon: "question",
            title: "確定刪除？",
            showDenyButton: true,
            showCancelButton: false,
            confirmButtonText: "確定",
            denyButtonText: "放棄"
        }).then((result) => {
            if (result.isConfirmed){
                // 使用ajax是為了讓資料送到後端執行
                $.ajax({
                    // 送去哪裡
                    url: "/admin/milestone/types/delete",
                    type: "post",
                    data: {
                        // 傳給後端的資料
                        id: id,
                        _token: "{{ csrf_token() }}"
                    },
                    // msg是接住後端回傳的內容
                    success: function(msg){
                        Swal.fire("已刪除!", "", "success");
                        /* btn: 傳進來的button
                           closest: 尋找最近符合條件的元素
                           tr: 表格中的列
                           [data-row]: 具有data-row屬性
                           tr[data-row]: 從目前元素(button)開始，往上尋找最近的一個符合條件的<tr>
                           remove()只會從畫面上移除這列，不會移除資料庫的，資料庫那邊要交由controller移除
                            */
                        // 整句意思：從按鈕往上，找到最近一個帶有 data-row 這個 HTML 屬性的 <tr>，並從畫面上移除
                        $(btn).closest("tr[data-row]").remove();
                    }
                });
            }
        });
    }
</script>

class MileStoneType extends Model
{
    public $timestamps = false;
    protected $table = "milestone_type";
    protected $primaryKey = 'id';
    protected $fillable = ["id", "label", "title"];

    public function milestone(): HasMany
    {
        // 每一個類別可能有多筆大事記
        return $this->hasMany(
            MileStone::class,
            "typeId", // 大事記資料表的外來鍵
            "id" // milestone_type的主鍵
        );
    }
}

MileStone：

// 大事記
class MileStone extends Model
{
    public $timestamps = false;
    protected $table = "milestone";
    protected $primaryKey = 'id';
    protected $fillable = ["id", "typeId", "years", "months", "title", "content", "createTime"];

    public function types(): BelongsTo
    {
        // typeId：milestone資料表的外來鍵
        // id：milestone_type的主鍵
        // 等同於 SELECT a.*, b.* FROM milestone a INNER JOIN milestone_type b ON a.typeId = b.id
        /*
        也可以用(DB要use Illuminate\Support\Facades\DB)
          DB::table("milestone as a")->selectRaw("a.*, b.*")->join("milestone_type as b")->get()
        或
          DB::table("milestone as a")->join("milestone_type as b")->selectRaw("a.*, b.*")->get()
        順序不影響
        DB：資料庫  table：資料表   join：inner join兩邊都有資料    selectRaw：查詢欄位 get：取全部資料
        selectRaw("a.*, b.*")也可以改用select("a.*", "b.*")
        */
        return $this->belongsTo(MileStoneType::class, "typeId", "id");
    }
}

/*
    加入middleware的寫法：
    1. 單一一條：路由後加->middelware("manager")

        });
    3. 用陣列一起設定middleware、網址前綴等等：
        Route::group(["middleware" => "manager", "prefix" => "admin"], function(){
            Route::...

middleware可以前往app/Http/Middleware/CheckManager.php查看（CheckManager是之前php artisan make:middleware CheckManager創建的）
啟用middleware，要先到app/bootstrap/app.php設定別名：
    ->withMiddleware(function (Middleware $middleware): void {
        $middleware->alias([
            "manager" => CheckManager::class
        ]);
    })


CheckManager.php裡設定：
class CheckManager
{
    public function handle(Request $request, Closure $next): Response
    {
        if(empty(session()->get("userId"))){
            return redirect("/admin");
        }
        return $next($request);
    }
}
Request $request：接收目前使用者的 HTTP 請求
Closure $next：接收一個可以讓請求繼續往下執行的函式
: Response：宣告這個方法會回傳 Response 類型的結果
$request：使用者送來的請求
$next：檢查通過後，放行到下一個處理階段

    public function store(Request $req){
        $content = $req->content;
        // 使用者輸入有可能會按enter換行
        if (!empty($content)){
            // str_replace(搜尋, 換成替代文字, 主題)
            // \n:(new line) 輸入框的換行(enter) 
            // 將輸入框的換行取代為網頁的換行
            $content = str_replace("\n", "<br/>", $content);
        }
        $ms = new MileStone();
        $ms->typeId = $req->typeId;
        $ms->years = $req->year;
        $ms->months = $req->month;
        $ms->title = $req->title;
        $ms->content = $content;
        $ms->point = $req->point;
        $ms->save();

        Session::flash("message", "已新增");
        return redirect("/admin/milestone/list");
    }

<script>
        // 傳入被點擊的按鈕，刪除成功後移除它所在的資料列
        function doDelete(id, btn, url) {
            Swal.fire({
                icon: "question",
                title: "確定刪除？",
                showDenyButton: true,
                showCancelButton: false,
                confirmButtonText: "確定",
                denyButtonText: "放棄"
            }).then((result) => {
                if (result.isConfirmed) {
                    // 使用ajax是為了讓資料送到後端執行
                    $.ajax({
                        // 送去哪裡
                        url: url,
                        type: "post",
                        data: {
                            // 傳給後端的資料
                            id: id,
                            _token: "{{ csrf_token() }}"
                        },
                        // msg是接住後端回傳的內容
                        success: function(msg) {
                            Swal.fire("已刪除!", "", "success");
                            /* btn: 傳進來的button
                                closest: 尋找最近符合條件的元素
                                tr: 表格中的列
                                [data-row]: 具有data-row屬性
                                tr[data-row]: 從目前元素(button)開始，往上尋找最近的一個符合條件的<tr>
                                remove()只會從畫面上移除這列，不會移除資料庫的，資料庫那邊要交由controller移除
                            */
                            // 整句意思：從按鈕往上，找到最近一個帶有 data-row 這個 HTML 屬性的 <tr>，並從畫面上移除
                            $(btn).closest("tr[data-row]").remove();
                        }
                    });
                }
            });
        }
    </script>
