<?php

namespace App\Models\Product;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;

class Product extends Model
{
    public $timestamps = false;
    protected $table = "product";
    protected $primaryKey = 'id';
    protected $fillable = ["id","label_id", "type_id", "product_name", "title", "product_no", "price", "sale_price", "qty", "content", "create_time"];

    public function types(): BelongsTo
    {
        return $this->belongsTo(ProductType::class, "type_id", "id");
    }

    public function labels(): BelongsTo
    {
        return $this->belongsTo(ProductLabel::class, "label_id", "id");
    }

    // 商品圖
    public function photos(): HasMany
    {
        return $this->hasMany(ProductPhoto::class, "productId", "id");
    }

    // 商品規格
    public function specs(): HasMany
    {
        return $this->hasMany(ProductSpec::class, "productId", "id");
    }

    // 一個商品有多張圖，要取第一張圖出來當封面
    public function firstPhoto()
    {
        return $this->photos->first();
    }
}

****建立ProductTypeController： php artisan make:controller Admin/Product/ProductTypeController
<?php

namespace App\Http\Controllers\Admin\Product;

use App\Http\Controllers\Controller;
use App\Models\Product\ProductType;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Session;

class ProductTypeController extends Controller
{
    public function list(){
        $list = ProductType::get();
        return view("admin.product.types.list", compact("list"));
    }

    public function create(){
        return view("admin.product.types.create");
    }

    public function store(Request $req){
        $types = new ProductType();
        $types->title = $req->title;
        $types->save();

        Session::flash("message", "已新增");
        return redirect("/admin/product/types/list");
    }

    public function edit(Request $req){
        $types = ProductType::find($req->id);
        return view("admin.product.types.edit", compact("types"));
    }

    public function update(Request $req){
        $types = ProductType::find($req->id);
        $types->title = $req->title;
        $types->update();

        Session::flash("message", "已修改");
        return redirect("/admin/product/types/list");
    }

    public function delete(Request $req){
        ProductType::destroy($req->id);
        Session::flash("message", "已刪除");
        return redirect("/admin/product/types/list");
    }
}

{
