<?php

namespace App\Models\Product;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;

class Product extends Model
{
    public $timestamps = false;
    protected $table = "product";
    protected $primaryKey = 'id';
    protected $fillable = ["id","label_id", "type_id", "product_name", "title", "product_no", "price", "sale_price", "qty", "content", "create_time"];

    public function types(): BelongsTo
    {
        return $this->belongsTo(ProductType::class, "typeId", "id");
    }

    public function labels(): BelongsTo
    {
        return $this->belongsTo(ProductLabel::class, "label_id", "id");
    }
}
